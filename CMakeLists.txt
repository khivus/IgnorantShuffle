cmake_minimum_required(VERSION 3.20)
project(IgnorantShuffle VERSION 0.11.0 LANGUAGES CXX)
set(IGNORANT_SHUFFLE_UPDATE_REPOSITORY "khivus/IgnorantShuffle" CACHE STRING "GitHub owner/repository for stable releases")
if(NOT IGNORANT_SHUFFLE_UPDATE_REPOSITORY MATCHES "^$|^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "Update repository must be blank or owner/repository")
endif()
configure_file(src/build_config.h.in generated/build_config.h @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(shuffle_core STATIC src/weights.cpp src/weight_storage.cpp)
find_package(Threads REQUIRED)
add_library(shuffle_diagnostics STATIC src/logging.cpp)
target_include_directories(shuffle_diagnostics PUBLIC src)
target_link_libraries(shuffle_diagnostics PUBLIC Threads::Threads)
target_include_directories(shuffle_core PUBLIC src)
if(MSVC)
  target_compile_options(shuffle_core PRIVATE /W4 /utf-8)
  target_compile_options(shuffle_diagnostics PRIVATE /W4 /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(shuffle_tests tests/core_tests.cpp)
  target_link_libraries(shuffle_tests PRIVATE shuffle_core)
  add_test(NAME shuffle_core COMMAND shuffle_tests)
  add_executable(storage_tests tests/storage_tests.cpp)
  target_link_libraries(storage_tests PRIVATE shuffle_core)
  add_test(NAME portable_storage COMMAND storage_tests)
  add_executable(logging_tests tests/logging_tests.cpp)
  target_link_libraries(logging_tests PRIVATE shuffle_diagnostics)
  add_test(NAME plugin_logging COMMAND logging_tests)
endif()
option(BUILD_AIMP_PLUGIN "Build the Windows AIMP DLL" ON)
set(AIMP_SDK_DIR "${CMAKE_SOURCE_DIR}/.deps/aimp_sdk_540" CACHE PATH "Extracted official AIMP 5.40 SDK directory")
if(BUILD_AIMP_PLUGIN AND WIN32)
  add_library(shuffle_updates STATIC src/update_model.cpp src/update_http.cpp src/update_client.cpp src/update_install.cpp)
  target_include_directories(shuffle_updates PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(shuffle_updates PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(shuffle_updates PRIVATE /W4 /utf-8)
  target_link_libraries(shuffle_updates PUBLIC winhttp bcrypt shell32 Threads::Threads)
  enable_language(RC)
  find_path(AIMP_INCLUDE_DIR apiPlaylists.h PATHS "${AIMP_SDK_DIR}/Sources/Cpp" "${AIMP_SDK_DIR}" NO_DEFAULT_PATH)
  if(NOT AIMP_INCLUDE_DIR)
    message(FATAL_ERROR "AIMP SDK headers are missing. Run tools/prepare-sdk.ps1 or set AIMP_SDK_DIR to an extracted AIMP 5.40 SDK. Use BUILD_AIMP_PLUGIN=OFF for core tests only.")
  endif()
  add_library(aimp_ignorant_shuffle SHARED src/plugin.cpp src/monitor.cpp src/track_table.cpp src/plugin.rc src/plugin.def)
  target_include_directories(aimp_ignorant_shuffle SYSTEM PRIVATE "${AIMP_INCLUDE_DIR}")
  target_compile_definitions(aimp_ignorant_shuffle PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(aimp_ignorant_shuffle PRIVATE shuffle_core shuffle_diagnostics shuffle_updates comctl32 comdlg32 ole32 uuid shell32 uxtheme)
  # Official headers contain ANSI copyright bytes. Our adapter uses ASCII source.
  target_compile_options(aimp_ignorant_shuffle PRIVATE /W4 /source-charset:windows-1252 /execution-charset:utf-8)
  set_target_properties(aimp_ignorant_shuffle PROPERTIES PREFIX "")
  if(BUILD_TESTING)
    add_executable(adapter_tests tests/adapter_tests.cpp src/monitor.cpp src/track_table.cpp src/plugin.rc)
    target_include_directories(adapter_tests SYSTEM PRIVATE "${AIMP_INCLUDE_DIR}")
    target_compile_definitions(adapter_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(adapter_tests PRIVATE /W4 /source-charset:windows-1252 /execution-charset:utf-8)
    target_link_libraries(adapter_tests PRIVATE shuffle_core shuffle_diagnostics shuffle_updates comctl32 comdlg32 ole32 uuid shell32 uxtheme)
    add_dependencies(adapter_tests aimp_ignorant_shuffle)
    add_test(NAME aimp_adapter COMMAND adapter_tests "$<TARGET_FILE:aimp_ignorant_shuffle>")
    add_executable(update_tests tests/update_tests.cpp)
    target_compile_options(update_tests PRIVATE /W4 /utf-8)
    target_link_libraries(update_tests PRIVATE shuffle_updates ws2_32)
    add_executable(update_install_probe WIN32 tests/install_probe.cpp)
    target_compile_options(update_install_probe PRIVATE /W4 /utf-8)
    target_link_libraries(update_install_probe PRIVATE shell32)
    add_dependencies(update_tests update_install_probe)
    add_test(NAME release_updates COMMAND update_tests "$<TARGET_FILE:update_install_probe>")
  endif()
endif()
